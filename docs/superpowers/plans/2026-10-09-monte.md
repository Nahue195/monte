`lib/dates.ts`:
```ts
const DAY_MS = 86_400_000;

const toTime = (iso: string) => Date.parse(`${iso}T00:00:00Z`);
const toIso = (time: number) => new Date(time).toISOString().slice(0, 10);
const weekdayOf = (iso: string) => new Date(toTime(iso)).getUTCDay();

export function localToday(now: Date = new Date()): string {
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())}`;
}

export function addDays(iso: string, days: number): string {
  return toIso(toTime(iso) + days * DAY_MS);
}

// The next given weekday strictly after `iso` (0 = Sunday).
export function nextWeekday(iso: string, weekday: number): string {
  const diff = (weekday - weekdayOf(iso) + 7) % 7;
  return addDays(iso, diff === 0 ? 7 : diff);
}

export function addBusinessDays(iso: string, n: number): string {
  let date = iso;
  for (let left = n; left > 0; ) {
    date = addDays(date, 1);
    const day = weekdayOf(date);
    if (day !== 0 && day !== 6) left--;
  }
  return date;
}
```

`lib/i18n.ts`:
```ts
export const locales = ["es", "en"] as const;

export type Locale = (typeof locales)[number];

export const defaultLocale: Locale = "es";

export function isLocale(value: string): value is Locale {
  return (locales as readonly string[]).includes(value);
}

// Parses an Accept-Language header and returns the best supported locale.
export function pickLocale(acceptLanguage: string | null | undefined): Locale {
  if (!acceptLanguage) return defaultLocale;

  const ranked = acceptLanguage
    .split(",")
    .map((part, index) => {
      const [tag = "", ...params] = part.split(";");
      const qParam = params.map((p) => p.trim()).find((p) => p.startsWith("q="));
      const quality = qParam ? Number(qParam.slice(2)) : 1;
      return {
        lang: tag.trim().toLowerCase().split("-")[0],
        quality: Number.isFinite(quality) ? quality : 0,
        index,
      };
    })
    .filter((entry) => entry.lang && entry.quality > 0)
    .sort((a, b) => b.quality - a.quality || a.index - b.index);

  for (const { lang } of ranked) {
    if (isLocale(lang)) return lang;
  }
  return defaultLocale;
}

// Same URL with the locale as path, keeping the query string (UTM and friends).
export function localizedUrl(requestUrl: string, locale: Locale): URL {
  const url = new URL(requestUrl);
  url.pathname = `/${locale}`;
  return url;
}

export function intlLocale(lang: Locale): string {
  return lang === "es" ? "es-AR" : "en-US";
}

// "/es/shop" → "/en/shop": routes are the same in both languages.
export function switchLocalePath(pathname: string, to: Locale): string {
  const segments = pathname.split("/").filter(Boolean);
  if (segments.length > 0 && isLocale(segments[0])) segments.shift();
  return `/${[to, ...segments].join("/")}`;
}
```

export function formatPrice(amount: number, lang: Locale): string {
  return new Intl.NumberFormat(intlLocale(lang), {
    style: "currency",
    currency: "ARS",
    maximumFractionDigits: 0,
  }).format(amount);

  return new Intl.DateTimeFormat(intlLocale(lang), {
    weekday: "long",
    day: "numeric",
    month: "long",
    timeZone: "UTC",
  }).format(new Date(`${iso}T00:00:00Z`));

export function fill(template: string, vars: Record<string, string | number>): string {
  return template.replace(/\{(\w+)\}/g, (match, key: string) => (key in vars ? String(vars[key]) : match));
}

export function levelWord(n: number): "low" | "mid" | "high" {
  if (n <= 2) return "low";

// Lowercase without diacritics: "Etiopía" → "etiopia".
export function fold(text: string): string {
  return text.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");

export type Term =
  | { kind: "note"; value: NoteTag }

import { fold } from "../text.ts";

export const MAX_QUERY_LENGTH = 200;

// "¿Algo frutal, menos de $15.000?" → ["algo", "frutal", "menos", "de", "15000"]
export function normalize(text: string): string[] {
  let s = fold(text.slice(0, MAX_QUERY_LENGTH)).replace(/\$/g, " ");
  for (let prev = ""; prev !== s; ) {
    prev = s;
    s = s.replace(/(\d)[.,](\d{3})(?!\d)/g, "$1$2");
  }
  s = s.replace(/(\d+)\s*(?:mil|k)\b/g, (_, n: string) => String(Number(n) * 1000));
  s = s.replace(/[^a-z0-9]+/g, " ").trim();
  return s ? s.split(" ") : [];
}

import { LEXICON, STOPWORDS } from "./lexicon.ts";
import type { Hit, Term } from "./types.ts";

const exact = new Map<string, Term>();
const singleWords: { word: string; term: Term }[] = [];
for (const { phrases, term } of LEXICON) {
  for (const phrase of phrases) {
    exact.set(phrase, term);
    if (!phrase.includes(" ")) singleWords.push({ word: phrase, term });
  }
}

// Damerau–Levenshtein, optimal string alignment variant (a swap of two letters costs 1).
export function editDistance(a: string, b: string): number {
  const d: number[][] = Array.from({ length: a.length + 1 }, (_, i) =>
    Array.from({ length: b.length + 1 }, (_, j) => (i === 0 ? j : j === 0 ? i : 0)),
  );
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      const cost = a[i - 1] === b[j - 1] ? 0 : 1;
      d[i][j] = Math.min(d[i - 1][j] + 1, d[i][j - 1] + 1, d[i - 1][j - 1] + cost);
      if (i > 1 && j > 1 && a[i - 1] === b[j - 2] && a[i - 2] === b[j - 1]) {
        d[i][j] = Math.min(d[i][j], d[i - 2][j - 2] + 1);
      }
    }
  }
  return d[a.length][b.length];
}

function allowedTypos(word: string): number {
  if (word.length >= 7) return 2;
  return word.length >= 4 ? 1 : 0;
}

function fuzzyWord(word: string): Term | undefined {
  const allowed = allowedTypos(word);
  if (allowed === 0 || STOPWORDS.has(word) || /\d/.test(word)) return undefined;
  let best: { term: Term; distance: number } | undefined;
  for (const candidate of singleWords) {
    if (Math.abs(candidate.word.length - word.length) > allowed) continue;
    const distance = editDistance(word, candidate.word);
    if (distance <= allowed && (!best || distance < best.distance)) best = { term: candidate.term, distance };
  }
  return best?.term;
}

// Longest phrase first (3, 2, 1 words); single words may carry a typo.
export function recognize(tokens: string[]): Hit[] {
  const hits: Hit[] = [];
  let i = 0;
  while (i < tokens.length) {
    let advanced = false;
    for (const len of [3, 2, 1]) {
      if (i + len > tokens.length) continue;
      const phrase = tokens.slice(i, i + len).join(" ");
      const term = exact.get(phrase) ?? (len === 1 ? fuzzyWord(phrase) : undefined);
      if (!term) continue;
      if (term.kind === "price-cue") {
        const next = tokens[i + len];
        if (next === undefined || !/^\d+$/.test(next)) continue;
        hits.push({ term: { kind: "price", max: Number(next) }, start: i, end: i + len + 1 });
        i += len + 1;
      } else {
        hits.push({ term, start: i, end: i + len });
        i += len;
      }
      advanced = true;
      break;
    }
    if (!advanced) i++;
  }
  return hits;
}

import { ROASTS } from "../catalog.ts";
import { normalize } from "./normalize.ts";
import { recognize } from "./recognize.ts";
import type { Query, Range, Term } from "./types.ts";

export const CHEAP_MAX = 13000;

const MILD_NEGATIONS = ["no muy", "no tan", "not very", "not too"];
const NEGATIONS = new Set(["no", "nada", "sin", "poco", "poca", "ni", "not", "without"]);
const INTENSIFIERS = new Set(["muy", "bien", "re", "super", "mas", "bastante", "very", "really", "more", "extra"]);

type Modifier = "mild-neg" | "neg" | "up" | null;

export function emptyQuery(): Query {
  return { notes: [], excludeNotes: [], methods: [], roasts: [], origins: [], levels: {}, decaf: false, recognized: 0 };
}

export function hasCriteria(q: Query): boolean {
  return (
    q.notes.length + q.excludeNotes.length + q.methods.length + q.roasts.length + q.origins.length > 0 ||
    Object.keys(q.levels).length > 0 ||
    q.maxPrice !== undefined ||
    q.decaf
  );
}

function modifierOf(before: string[]): Modifier {
  const joined = before.join(" ");
  if (MILD_NEGATIONS.some((m) => joined.endsWith(m))) return "mild-neg";
  if (before.some((w) => NEGATIONS.has(w))) return "neg";
  if (before.some((w) => INTENSIFIERS.has(w))) return "up";
  return null;
}

const add = <T>(list: T[], value: T) => {
  if (!list.includes(value)) list.push(value);
};

function levelRange(dir: "up" | "down", mod: Modifier): Range {
  if (dir === "down") return { max: 2 };
  if (mod === "mild-neg") return { max: 3 };
  if (mod === "neg") return { max: 2 };
  return mod === "up" ? { min: 4 } : { min: 3 };
}

function apply(q: Query, term: Term, mod: Modifier) {
  const negated = mod === "neg" || mod === "mild-neg";
  switch (term.kind) {
    case "note":
      if (negated) {
        add(q.excludeNotes, term.value);
        q.notes = q.notes.filter((n) => n !== term.value);
      } else if (!q.excludeNotes.includes(term.value)) {
        add(q.notes, term.value);
      }
      break;
    case "method":
      if (!negated) add(q.methods, term.value);
      break;
    case "roast":
      for (const roast of negated ? ROASTS.filter((r) => r !== term.value) : [term.value]) add(q.roasts, roast);
      break;
    case "origin":
      if (!negated) for (const origin of term.value) add(q.origins, origin);
      break;
    case "level":
      q.levels[term.attr] = levelRange(term.dir, mod);
      break;
    case "decaf":
      q.decaf = true;
      break;
    case "cheap":
      q.maxPrice = Math.min(q.maxPrice ?? Infinity, CHEAP_MAX);
      break;
    case "price":
      q.maxPrice = Math.min(q.maxPrice ?? Infinity, term.max);
      break;
    case "price-cue":
      break;
  }
}

export function interpret(text: string): Query {
  const tokens = normalize(text);
  const hits = recognize(tokens);
  const q = emptyQuery();
  let previousEnd = 0;
  for (const hit of hits) {
    const before = tokens.slice(Math.max(previousEnd, hit.start - 2), hit.start);
    apply(q, hit.term, modifierOf(before));
    previousEnd = hit.end;
  }
  q.recognized = hits.length;
  return q;
}

import type { Query, Range } from "./types.ts";

export const NOTE_FAMILY: Record<NoteTag, NoteTag[]> = {
  frutal: ["frutal", "citrico", "frutos-rojos"],
  citrico: ["citrico"],
  "frutos-rojos": ["frutos-rojos"],
  chocolate: ["chocolate"],
  caramelo: ["caramelo"],
  floral: ["floral"],
  nueces: ["nueces"],
  especiado: ["especiado"],

export type MatchKey =
  | { type: "notes"; values: NoteTag[] }
  | { type: "method"; values: Method[] }
  | { type: "roast"; values: Roast[] }
  | { type: "origin"; values: string[] }
  | { type: "level"; attr: Attr; range: Range }
  | { type: "exclude"; value: NoteTag }
  | { type: "price"; max: number }
  | { type: "decaf" };

export type Match = { key: MatchKey; ok: boolean };
export type Result = { coffee: Coffee; score: number; matches: Match[]; close: boolean };
export type SearchStatus = "all" | "matched" | "not-understood" | "none-pass";
export type SearchResult = { status: SearchStatus; results: Result[] };

type Soft = { key: MatchKey; weight: number; score: (c: Coffee) => number };
type Hard = { key: MatchKey; pass: (c: Coffee) => boolean };

export function levelScore(value: number, range: Range): number {
  const distance =
    range.min !== undefined && value < range.min
      ? range.min - value
      : range.max !== undefined && value > range.max
        ? value - range.max
        : 0;
  if (distance === 0) return 1;
  return distance === 1 ? 0.5 : 0;
}

function criteria(q: Query): { soft: Soft[]; hard: Hard[] } {
  const soft: Soft[] = [];
  const hard: Hard[] = [];
  if (q.notes.length) {
    const tags = family(q.notes);
    soft.push({ key: { type: "notes", values: q.notes }, weight: 3, score: (c) => (hasAny(c, tags) ? 1 : 0) });
  }
  if (q.methods.length) {
    soft.push({ key: { type: "method", values: q.methods }, weight: 3, score: (c) => (c.methods.some((m) => q.methods.includes(m)) ? 1 : 0) });
  }
  if (q.roasts.length) {
    soft.push({ key: { type: "roast", values: q.roasts }, weight: 2, score: (c) => (q.roasts.includes(c.roast) ? 1 : 0) });
  }
  if (q.origins.length) {
    soft.push({ key: { type: "origin", values: q.origins }, weight: 2, score: (c) => (q.origins.includes(countryKey(c)) ? 1 : 0) });
  }
  for (const attr of ATTRS) {
    const range = q.levels[attr];
    if (range) soft.push({ key: { type: "level", attr, range }, weight: 2, score: (c) => levelScore(c[attr], range) });
  }
  for (const note of q.excludeNotes) {
    const tags = NOTE_FAMILY[note];
    hard.push({ key: { type: "exclude", value: note }, pass: (c) => !hasAny(c, tags) });
  }
  if (q.maxPrice !== undefined) {
    const max = q.maxPrice;
    hard.push({ key: { type: "price", max }, pass: (c) => c.prices["250"] <= max });
  }
  if (q.decaf) hard.push({ key: { type: "decaf" }, pass: (c) => c.decaf });
  return { soft, hard };
}

const byName = (a: Coffee, b: Coffee) => a.name.localeCompare(b.name, "es");
const byScoreThenName = (a: Result, b: Result) => b.score - a.score || byName(a.coffee, b.coffee);

export function rank(coffees: Coffee[], q: Query, hasText: boolean): SearchResult {
  const { soft, hard } = criteria(q);

  if (soft.length === 0 && hard.length === 0) {
    const ordered = [...coffees].sort((a, b) => Number(Boolean(b.featured)) - Number(Boolean(a.featured)) || byName(a, b));
    return {
      status: hasText ? "not-understood" : "all",
      results: ordered.map((coffee) => ({ coffee, score: 0, matches: [], close: false })),
    };
  }

  const possible = soft.reduce((sum, s) => sum + s.weight, 0);
  const evaluated = coffees.map((coffee) => {
    const scores = soft.map((s) => s.score(coffee));
    const got = soft.reduce((sum, s, i) => sum + s.weight * scores[i], 0);
    const passes = hard.map((h) => h.pass(coffee));
    const matches: Match[] = [
      ...soft.map((s, i) => ({ key: s.key, ok: scores[i] === 1 })),
      ...hard.map((h, i) => ({ key: h.key, ok: passes[i] })),
    ];
    const result: Result = { coffee, score: possible ? got / possible : 1, matches, close: false };
    return { result, passesAll: passes.every(Boolean) };
  });
  evaluated.sort((a, b) => byScoreThenName(a.result, b.result));

  const passing = evaluated.filter((e) => e.passesAll).map((e) => e.result);
  if (passing.length === 0) {
    return { status: "none-pass", results: evaluated.slice(0, 3).map((e) => e.result) };
  }

  const hits = passing.filter((r) => r.score > 0);
  const fill = passing
    .filter((r) => r.score === 0)
    .slice(0, Math.max(0, 3 - hits.length))
    .map((r) => ({ ...r, close: true }));
  return { status: "matched", results: [...hits, ...fill] };
}

export type StorageLike = Pick<Storage, "getItem" | "setItem">;

export type Persisted<T> = {
  get(): T;
  set(value: T): void;
  subscribe(listener: () => void): () => void;
  refresh(): void;
  serverSnapshot(): T;
};

function browserStorage(): StorageLike | undefined {
  try {
    return typeof window === "undefined" ? undefined : window.localStorage;
  } catch {
    return undefined;
  }
}

// External store for useSyncExternalStore: validated reads, stable snapshots,
// keeps working in memory when storage is blocked, and follows other tabs.
export function createPersisted<T>(
  key: string,
  parse: (raw: unknown) => T,
  empty: T,
  getStorage: () => StorageLike | undefined = browserStorage,
): Persisted<T> {
  let cache: { value: T } | undefined;
  const listeners = new Set<() => void>();
  const notify = () => listeners.forEach((listener) => listener());

  const read = (): T => {
    try {
      const raw = getStorage()?.getItem(key);
      return raw == null ? empty : parse(JSON.parse(raw));
    } catch {
      return empty;
    }
  };

  const store: Persisted<T> = {
    get() {
      cache ??= { value: read() };
      return cache.value;
    },
    set(value) {
      cache = { value };
      try {
        getStorage()?.setItem(key, JSON.stringify(value));
      } catch {
        // Storage full or blocked: keep the value in memory.
      }
      notify();
    },
    refresh() {
      cache = undefined;
      notify();
    },
    subscribe(listener) {
      listeners.add(listener);
      const onStorage = (event: StorageEvent) => {
        if (event.key === key) store.refresh();
      };
      if (typeof window !== "undefined") window.addEventListener("storage", onStorage);
      return () => {
        listeners.delete(listener);
        if (typeof window !== "undefined") window.removeEventListener("storage", onStorage);
      };
    },
    serverSnapshot: () => empty,
  };
  return store;
}

import { SIZES, type Coffee, type Size } from "./catalog.ts";

export const CART_KEY = "monte:cart:v1";
export const GRINDS = ["grano", "filtro", "espresso", "prensa"] as const;
export const MAX_QTY = 10;
export const FREE_SHIPPING_FROM = 40000;
export const SHIPPING_COST = 4500;

export type Grind = (typeof GRINDS)[number];
export type Delivery = "envio" | "retiro";
export type CartLine = { slug: string; grind: Grind; size: Size; qty: number };
export type PricedLine = CartLine & { coffee: Coffee; unitPrice: number; total: number };
export type PricedCart = { lines: PricedLine[]; subtotal: number; shipping: number; total: number; freeShippingGap: number };

const clampQty = (qty: number) => Math.min(MAX_QTY, Math.max(1, Math.trunc(qty)));

export function lineKey(line: Pick<CartLine, "slug" | "grind" | "size">): string {
  return `${line.slug}|${line.grind}|${line.size}`;
}

export function addLine(lines: CartLine[], line: CartLine): CartLine[] {
  const key = lineKey(line);
  if (lines.some((l) => lineKey(l) === key)) {
    return lines.map((l) => (lineKey(l) === key ? { ...l, qty: clampQty(l.qty + line.qty) } : l));
  }
  return [...lines, { ...line, qty: clampQty(line.qty) }];
}

export function removeLine(lines: CartLine[], key: string): CartLine[] {
  return lines.filter((l) => lineKey(l) !== key);
}

export function setQty(lines: CartLine[], key: string, qty: number): CartLine[] {
  if (qty < 1) return removeLine(lines, key);
  return lines.map((l) => (lineKey(l) === key ? { ...l, qty: clampQty(qty) } : l));
}

export function cartCount(lines: CartLine[]): number {
  return lines.reduce((sum, l) => sum + l.qty, 0);
}

// Prices always come from the catalog, never from what was stored.
export function priceCart(lines: CartLine[], catalog: Coffee[], delivery: Delivery = "envio"): PricedCart {
  const priced = lines.flatMap((line): PricedLine[] => {
    const coffee = catalog.find((c) => c.slug === line.slug);
    if (!coffee) return [];
    const unitPrice = coffee.prices[line.size];
    return [{ ...line, coffee, unitPrice, total: unitPrice * line.qty }];
  });
  const subtotal = priced.reduce((sum, l) => sum + l.total, 0);
  const free = subtotal >= FREE_SHIPPING_FROM;
  const shipping = priced.length === 0 || delivery === "retiro" || free ? 0 : SHIPPING_COST;
  return {
    lines: priced,
    subtotal,
    shipping,
    total: subtotal + shipping,
    freeShippingGap: free ? 0 : FREE_SHIPPING_FROM - subtotal,
  };
}

export function parseCart(raw: unknown, catalog: Coffee[]): CartLine[] {
  if (!Array.isArray(raw)) return [];
  let lines: CartLine[] = [];
  for (const item of raw) {
    if (!item || typeof item !== "object") continue;
    const { slug, grind, size, qty } = item as Record<string, unknown>;
    if (typeof slug !== "string" || !catalog.some((c) => c.slug === slug)) continue;
    if (!GRINDS.includes(grind as Grind) || !SIZES.includes(size as Size)) continue;
    if (typeof qty !== "number" || !Number.isFinite(qty) || qty < 1) continue;
    lines = addLine(lines, { slug, grind: grind as Grind, size: size as Size, qty });
  }
  return lines;
}

export const FIELD_ORDER = ["email", "phone", "name", "address", "city", "province", "zip", "card", "expiry", "cvv"] as const;

export type Field = (typeof FIELD_ORDER)[number];
export type CheckoutForm = Record<Field, string> & { delivery: Delivery };
export type FieldError = "required" | "invalid" | "expired";
export type FormErrors = Partial<Record<Field, FieldError>>;

const SHIPPING_ONLY: Field[] = ["name", "address", "city", "province", "zip"];

const lengthBetween = (value: string, min: number, max: number) => value.length >= min && value.length <= max;

export function luhn(digits: string): boolean {
  if (!/^\d+$/.test(digits)) return false;
  let sum = 0;
  let double = false;
  for (let i = digits.length - 1; i >= 0; i--) {
    let d = Number(digits[i]);
    if (double) {
      d *= 2;
      if (d > 9) d -= 9;
    }
    sum += d;
    double = !double;
  }
  return sum % 10 === 0;
}

function checkExpiry(value: string, today: string): FieldError | null {
  const match = /^(\d{2})\s*\/\s*(\d{2})$/.exec(value);
  if (!match) return "invalid";
  const month = Number(match[1]);
  if (month < 1 || month > 12) return "invalid";
  const year = 2000 + Number(match[2]);
  const [thisYear, thisMonth] = today.split("-").map(Number);
  return year < thisYear || (year === thisYear && month < thisMonth) ? "expired" : null;
}

export function validateField(field: Field, form: CheckoutForm, today: string): FieldError | null {
  if (form.delivery === "retiro" && SHIPPING_ONLY.includes(field)) return null;
  const value = form[field].trim();
  if (!value) return "required";
  switch (field) {
    case "email":
      return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(value) ? null : "invalid";
    case "phone":
      return /^\d{8,15}$/.test(value.replace(/[\s\-+().]/g, "")) ? null : "invalid";
    case "name":
    case "city":
      return lengthBetween(value, 2, 60) ? null : "invalid";
    case "address":
      return lengthBetween(value, 4, 100) ? null : "invalid";
    case "province":
      return PROVINCES.includes(value) ? null : "invalid";
    case "zip":
      return /^(\d{4}|[a-z]\d{4}[a-z]{3})$/i.test(value) ? null : "invalid";
    case "card": {
      const digits = value.replace(/[\s-]/g, "");
      return /^\d{13,19}$/.test(digits) && luhn(digits) ? null : "invalid";
    }
    case "expiry":
      return checkExpiry(value, today);
    case "cvv":
      return /^\d{3,4}$/.test(value) ? null : "invalid";
  }
  return null;
}

export function validateForm(form: CheckoutForm, today: string): FormErrors {
  const errors: FormErrors = {};
  for (const field of FIELD_ORDER) {
    const error = validateField(field, form, today);
    if (error) errors[field] = error;
  }
  return errors;
}

  const l = raw as Record<string, unknown>;

    typeof l.slug === "string" &&
    typeof l.name === "string" &&
    GRINDS.includes(l.grind as Grind) &&

    [l.qty, l.unitPrice, l.total].every(isNumber)
  );
}

export function parseOrder(raw: unknown): Order | null {
  if (!raw || typeof raw !== "object") return null;
  const o = raw as Record<string, unknown>;
  if (typeof o.id !== "string" || !/^MNT-[A-Z2-9]{6}$/.test(o.id)) return null;
  if (typeof o.date !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(o.date)) return null;

    const from = addBusinessDays(roast, 1);

// WCAG 2 contrast helpers, used to keep the theme tokens at AA.
export function parseTokens(css: string, selector: string): Record<string, string> {
  const start = css.indexOf(`${selector} {`);
  if (start < 0) return {};
  const block = css.slice(start, css.indexOf("}", start));
  return Object.fromEntries([...block.matchAll(/(--[\w-]+):\s*(#[0-9a-fA-F]{6})/g)].map((m) => [m[1], m[2]]));
}

function luminance(hex: string): number {
  const [r, g, b] = [1, 3, 5].map((i) => {
    const c = parseInt(hex.slice(i, i + 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}

export function contrastRatio(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return Math.round(((hi + 0.05) / (lo + 0.05)) * 100) / 100;
}

"use client";

import { createContext, useCallback, useContext, useMemo, useState, type ReactNode } from "react";

type CartUi = {
  isOpen: boolean;
  openCart: () => void;
  closeCart: () => void;
  announce: (message: string) => void;
};

const CartUiContext = createContext<CartUi | null>(null);

export function CartProvider({ children }: { children: ReactNode }) {
  const [isOpen, setOpen] = useState(false);
  const [message, setMessage] = useState("");
  const openCart = useCallback(() => setOpen(true), []);
  const closeCart = useCallback(() => setOpen(false), []);
  // Screen readers skip a repeated identical message, so alternate a trailing no-break space.
  const announce = useCallback((text: string) => setMessage((prev) => (prev === text ? `${text} ` : text)), []);
  const value = useMemo(() => ({ isOpen, openCart, closeCart, announce }), [isOpen, openCart, closeCart, announce]);

  return (
    <CartUiContext.Provider value={value}>
      {children}
      <p role="status" aria-live="polite" className="sr-only">
        {message}
      </p>
    </CartUiContext.Provider>
  );
}

export function useCartUi(): CartUi {
  const ctx = useContext(CartUiContext);
  if (!ctx) throw new Error("useCartUi must be used inside CartProvider");
  return ctx;
}

"use client";

import { usePathname } from "next/navigation";
import { switchLocalePath, type Locale } from "@/lib/i18n";

type Props = { lang: Locale; label: string };

// Keeps the current route and query (search, filters) when switching language.
export function LocaleSwitch({ lang, label }: Props) {
  const pathname = usePathname();
  const other: Locale = lang === "es" ? "en" : "es";
  const href = switchLocalePath(pathname, other);

  return (
    <a
      href={href}
      hrefLang={other}
      lang={other}
      aria-label={label}
      onClick={(event) => {
        const rest = window.location.search + window.location.hash;
        if (!rest) return;
        event.preventDefault();
        window.location.assign(href + rest);
      }}
      className="inline-flex h-10 items-center rounded-full px-2 text-xs font-semibold uppercase tracking-widest text-muted transition-colors hover:text-ink"
    >
      {other}
    </a>
  );
}

import type { Dictionary } from "../dictionaries/es.ts";
import type { Coffee } from "./catalog.ts";
import { fill, formatPrice, levelWord } from "./format.ts";
import type { Locale } from "./i18n.ts";
import type { Match } from "./search/rank.ts";

export type Labels = Dictionary["labels"];

const join = (parts: string[]) => parts.join(" / ");

// Text for one "✓ / ✗" chip under a search result.
export function matchLabel(match: Match, coffee: Coffee, labels: Labels, lang: Locale): string {
  const { key } = match;
  switch (key.type) {
    case "notes":
      return join(key.values.map((v) => labels.notes[v]));
    case "method":
      return join(key.values.map((v) => labels.methods[v]));
    case "roast":
      return join(key.values.map((v) => labels.roasts[v]));
    case "origin":
      return join(key.values.map((v) => (labels.origins as Record<string, string>)[v] ?? v));
    case "level":
      return `${labels.attrs[key.attr]} ${labels.levels[key.attr][levelWord(coffee[key.attr])]}`;
    case "exclude":
      return fill(labels.without, { note: labels.notes[key.value] });
    case "price":
      return fill(labels.upTo, { price: formatPrice(key.max, lang) });
    case "decaf":
      return labels.decaf;
  }
}

"use client";

import { useLayoutEffect, useRef } from "react";

// FLIP: when the order of `keys` changes, each element glides from its old spot.
export function useFlip(keys: string[]) {
  const nodes = useRef(new Map<string, HTMLElement>());
  const rects = useRef(new Map<string, DOMRect>());
  const signature = keys.join("|");

  useLayoutEffect(() => {
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    for (const [key, el] of nodes.current) {
      const next = el.getBoundingClientRect();
      const prev = rects.current.get(key);
      if (prev && !reduce) {
        const dx = prev.left - next.left;
        const dy = prev.top - next.top;
        if (dx || dy) {
          el.animate([{ transform: `translate(${dx}px, ${dy}px)` }, { transform: "none" }], {
            duration: 280,
            easing: "cubic-bezier(0.22, 1, 0.36, 1)",
          });
        }
      }
      rects.current.set(key, next);
    }
  }, [signature]);

  return (key: string) => (el: HTMLElement | null) => {
    if (el) nodes.current.set(key, el);
    else nodes.current.delete(key);
  };
}

"use client";

import { useCallback, useDeferredValue, useEffect, useMemo, useState } from "react";
import { usePathname, useRouter, useSearchParams } from "next/navigation";
import { Search, X } from "lucide-react";
import { COFFEES } from "@/data/coffees";
import type { Dictionary } from "@/dictionaries";
import { METHODS, ROASTS, type Method, type Roast } from "@/lib/catalog";
import { fill } from "@/lib/format";
import type { Locale } from "@/lib/i18n";
import { search } from "@/lib/search";
import { MAX_QUERY_LENGTH } from "@/lib/search/normalize";
import { ExampleChips } from "./ExampleChips";
import { ResultCard } from "./ResultCard";
import { useFlip } from "./useFlip";

type Props = { lang: Locale; dict: Pick<Dictionary, "search" | "shop" | "labels" | "product" | "cart"> };

const asMethod = (v: string | null) => (METHODS as readonly string[]).includes(v ?? "") ? (v as Method) : null;
const asRoast = (v: string | null) => (ROASTS as readonly string[]).includes(v ?? "") ? (v as Roast) : null;

export function ShopView({ lang, dict }: Props) {
  const params = useSearchParams();
  const router = useRouter();
  const pathname = usePathname();
  const [text, setText] = useState(() => params.get("q") ?? "");
  const method = asMethod(params.get("m"));
  const roast = asRoast(params.get("r"));
  const query = useDeferredValue(text);
  const result = useMemo(() => search(COFFEES, query, { method, roast }), [query, method, roast]);
  const flip = useFlip(result.results.map((r) => r.coffee.slug));

  const writeUrl = useCallback(
    (changes: Record<string, string | null>) => {
      const next = new URLSearchParams(window.location.search);
      for (const [key, value] of Object.entries(changes)) {
        if (value) next.set(key, value);
        else next.delete(key);
      }
      const qs = next.toString();
      router.replace(qs ? `${pathname}?${qs}` : pathname, { scroll: false });
    },
    [pathname, router],
  );

  // Keep the URL shareable without rewriting it on every keystroke.
  useEffect(() => {
    const id = setTimeout(() => writeUrl({ q: text.trim() || null }), 300);
    return () => clearTimeout(id);
  }, [text, writeUrl]);

  const n = result.results.length;
  const statusText =
    result.status === "all"
      ? fill(dict.shop.all, { n })
      : result.status === "matched"
        ? `${fill(dict.shop.results, { n })} · ${dict.shop.sortedBy}`
        : result.status === "none-pass"
          ? dict.shop.nonePass
          : dict.shop.notUnderstood;

  return (
    <div>
      <div role="search" className="flex items-center gap-2 rounded-full border border-line bg-panel py-1.5 pl-4 pr-1.5">
        <Search size={18} aria-hidden className="shrink-0 text-muted" />
        <input
          type="search"
          value={text}
          onChange={(e) => setText(e.target.value.slice(0, MAX_QUERY_LENGTH))}
          maxLength={MAX_QUERY_LENGTH}
          aria-label={dict.search.label}
          placeholder={dict.search.placeholder}
          autoComplete="off"
          className="min-w-0 flex-1 bg-transparent py-2.5 text-base text-ink outline-none placeholder:text-muted"
        />
        {text && (
          <button
            type="button"
            onClick={() => setText("")}
            aria-label={dict.search.clear}
            className="inline-flex h-10 w-10 shrink-0 items-center justify-center rounded-full text-muted hover:text-ink"
          >
            <X size={18} aria-hidden />
          </button>
        )}
      </div>

      <div className="mt-4 flex flex-col gap-3 sm:flex-row sm:flex-wrap sm:gap-6">
        <FilterGroup
          label={dict.shop.method}
          anyLabel={dict.shop.any}
          options={METHODS.map((m) => ({ value: m, label: dict.labels.methods[m] }))}
          current={method}
          onChange={(value) => writeUrl({ m: value })}
        />
        <FilterGroup
          label={dict.shop.roast}
          anyLabel={dict.shop.any}
          options={ROASTS.map((r) => ({ value: r, label: dict.labels.roasts[r] }))}
          current={roast}
          onChange={(value) => writeUrl({ r: value })}
        />
      </div>

      <p role="status" aria-live="polite" className="mt-6 text-sm text-muted" data-testid="results-status">
        {statusText}
      </p>
      {result.status === "not-understood" && (
        <ExampleChips lang={lang} label={dict.search.examplesLabel} examples={dict.search.examples} onPick={setText} />
      )}

      <ul className="mt-6 grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
        {result.results.map((r) => (
          <li key={r.coffee.slug} ref={flip(r.coffee.slug)}>
            <ResultCard result={r} lang={lang} dict={dict} />
          </li>
        ))}
      </ul>
    </div>
  );
}

type FilterGroupProps = {
  label: string;
  anyLabel: string;
  options: { value: string; label: string }[];
  current: string | null;
  onChange: (value: string | null) => void;
};

function FilterGroup({ label, anyLabel, options, current, onChange }: FilterGroupProps) {
  return (
    <div role="group" aria-label={label} className="flex flex-wrap items-center gap-2">
      <span className="text-xs uppercase tracking-widest text-muted">{label}</span>
      <button type="button" className="chip" aria-pressed={current === null} onClick={() => onChange(null)}>
        {anyLabel}
      </button>
      {options.map((option) => (
        <button
          key={option.value}
          type="button"
          className="chip"
          aria-pressed={current === option.value}
          onClick={() => onChange(current === option.value ? null : option.value)}
        >
          {option.label}
        </button>
      ))}
    </div>
  );
}

"use client";

import { useEffect, useRef, type ReactNode } from "react";

type Props = { open: boolean; onClose: () => void; labelledBy: string; children: ReactNode };

// Native <dialog>: focus trap, Escape and focus return come from the browser.
export function Dialog({ open, onClose, labelledBy, children }: Props) {
  const ref = useRef<HTMLDialogElement>(null);

  useEffect(() => {
    const dialog = ref.current;
    if (!dialog) return;
    if (open && !dialog.open) dialog.showModal();
    if (!open && dialog.open) dialog.close();
  }, [open]);

  return (
    <dialog
      ref={ref}
      aria-labelledby={labelledBy}
      className="sheet"
      onClose={onClose}
      onClick={(event) => {
        if (event.target === event.currentTarget) onClose();
      }}
    >
      {children}
    </dialog>
  );
}

    // Once a field shows an error, re-check it while typing so the message clears as soon as it's fixed.
    if (errors[field]) recheck(field, next);
  };

  const onDelivery = (delivery: Delivery) => setForm((prev) => ({ ...prev, delivery }));

  const fieldProps = (field: Field) => ({
    id: field,
    label: checkout.fields[field],
    value: form[field],

    onBlur: () => recheck(field, form),
  });

    event.preventDefault();
    if (busy.current) return;

    setErrors(found);
    setAttempted(true);
    const first = FIELD_ORDER.find((field) => found[field]);
    if (first) {
      document.getElementById(`f-${first}`)?.focus();
      return;
    }
    busy.current = true;
    const snapshot = priceCart(cartStore.get(), COFFEES, form.delivery);
    setFrozen(snapshot);
    await new Promise((resolve) => setTimeout(resolve, PROCESSING_MS));
    orderStore.set(buildOrder(snapshot, form.delivery, makeOrderId(), localToday()));
    cartActions.clear();
    router.push(`/${lang}/order`);
  }

  const hasErrors = Object.values(errors).some(Boolean);

# Monte

A specialty coffee shop that doesn't exist: brand, store and a search box that understands what you mean. A concept project by Nahuel Martinez.

## The idea

Most coffee shops make you pick by origin names you don't know. Monte lets you say what you like ("something fruity for pour over, not too acidic") and explains why each coffee fits.

## What it does

- **Natural-language search** in Spanish and English, with no AI: a hand-built interpreter.
- **Catalog of 12 coffees** with flavor profile, origin, process and brew methods.
- **Cart** that survives reloads and syncs across tabs, with free-shipping progress.
- **Simulated checkout** with inline validation (Luhn, Argentine postal codes) and an order confirmation. Nothing is charged and no data leaves the browser.

## How the search works

1. **Normalize**: lowercase, strip accents, unify prices ("15 mil", "$15.000", "15k" → 15000).
2. **Recognize**: match a bilingual lexicon, longest phrase first, tolerating typos (edit distance with transpositions, so "frutla" → "frutal").
3. **Modify**: read up to two words before each match for negation and intensity ("no muy ácido" → acidity ≤ 3, "sin chocolate" → exclude).
4. **Rank**: weighted score with hard filters (price, decaf, exclusions), partial credit for near misses, and a ✓ / ✗ explanation per result.

Every step is a pure function with table-driven tests.

## Decisions

- **No backend.** Everything is static and deploys for free; the cart and the last order live in `localStorage` behind a small external store with an in-memory fallback.
- **No AI.** The search is predictable, instant and explainable, and costs nothing per query.
- **No photos.** Coffee bags are drawn in CSS, so the brand stays consistent and light.
- **Accessible by default.** AA contrast is checked by a test, the cart is a native `<dialog>`, and result counts are announced to screen readers.

## Stack

Next.js 16 (App Router), React 19, TypeScript, Tailwind CSS 4, Node's built-in test runner.

## Run it locally

```bash
npm install
npm run dev     # http://localhost:3000
npm test        # unit tests
npm run lint
npm run build
```

Optional: set `NEXT_PUBLIC_PORTFOLIO_URL` to point the footer link to the portfolio.
